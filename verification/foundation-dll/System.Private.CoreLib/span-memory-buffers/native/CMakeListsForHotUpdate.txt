# Per-family hotupdate test from CodeGen-generated C++
# The patch file (native-aot.patch.generated.cpp) has RunNativeAot
# renamed to RunNativeAot_span_memory_buffers to avoid symbol collision with
# the host file's RunNativeAot.

add_executable(chaos_hotupdate_span_memory_buffers
    hotupdate/HotUpdateTest.cpp
    genuine/generated/native-aot.generated.cpp
    patch/native-aot.patch.generated.cpp
)
target_compile_features(chaos_hotupdate_span_memory_buffers PRIVATE cxx_std_17)
target_compile_definitions(chaos_hotupdate_span_memory_buffers PRIVATE CHAOS_RUNTIME_ABI_STATIC)
target_include_directories(chaos_hotupdate_span_memory_buffers PRIVATE
    ${CMAKE_SOURCE_DIR}/src/native/common
    ${CMAKE_SOURCE_DIR}/src/native/runtime-core
    ${CMAKE_SOURCE_DIR}/src/native/bootstrap
    ${CMAKE_SOURCE_DIR}/contracts/native/v0
)
target_link_libraries(chaos_hotupdate_span_memory_buffers PRIVATE
    chaos_hot_update
    chaos_bootstrap
    chaos_runtime_core
)
